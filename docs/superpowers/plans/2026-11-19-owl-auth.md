# Owl Real Login (Build Step 5) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the pretend login with real accounts: sign-up, login, logout and "who am I" on the backend, protected by a session cookie, CSRF checks and rate limits, and connect the frontend's auth pages and route protection to it.

**Architecture:** A new `app/auth/` package in the backend holds one small file per job (password hashing, session tokens, rate limiting, CSRF, request/response shapes, the "current user" dependency, routes, the admin reset). The frontend gains a tiny `apiFetch` client, TanStack Query for server data, auth hooks, and two route guards (`RequireAuth`, `GuestOnly`). Vite forwards `/api` to the backend, so the browser sees one origin and the cookie just works.

**Tech Stack:** Backend: FastAPI, SQLAlchemy, pwdlib (Argon2), PyJWT, pytest. Frontend: React, TanStack Query, React Router, React Hook Form + Zod, Vitest + React Testing Library.

**Spec:** `docs/superpowers/specs/2026-11-19-owl-rag-mvp-design.md` (sections 3, 5, 8, 9, 10, 12).

**Learner mode (owner's standing request):** Before writing or changing any file, show the file map for the task, explain the code in plain language (including the ideas listed under "Ideas"), then write it, then have the owner run and check it.

## Global Constraints

- Backend commands run from `backend/` with `uv run ...`; frontend commands from `frontend/` with `npm ...`. Shell: Git Bash.
- Work on a branch `feature/auth`, merged into `main` at the end.
- Passwords: 10–128 characters, hashed with **Argon2**; the plain password is never stored or logged (spec 5).
- Emails are stored trimmed and lowercase.
- Session: a signed JWT in an **httpOnly, SameSite=Lax** cookie named `owl_session`, valid **7 days**, `Secure` when `FRONTEND_URL` starts with `https://`. No refresh tokens. The token carries `token_version`; a mismatch means "logged out" (spec 5).
- Sign-up logs the user in automatically (spec 5).
- CSRF: every request except `GET`, `HEAD`, `OPTIONS` must carry an `Origin` header equal to `FRONTEND_URL`, otherwise `403` (spec 5).
- Rate limits are in memory (one backend instance, spec 13): login 20 per 15 minutes per IP and 10 per 15 minutes per email; sign-up 10 per hour per IP. Exceeded → `429`. **Deploy note (build step 9):** behind Vercel's `/api` forwarding every request arrives from the proxy's address, so `client_ip` must then read the forwarded client address (`X-Forwarded-For` from the trusted proxy) or the per-IP limits apply to everyone together.
- Error messages are plain English and returned as `{"detail": "..."}`. Login uses one message for "unknown email" and "wrong password": **"Wrong email or password."** Known MVP gap: sign-up says "An account with this email already exists." (spec 5).
- `GET /api/config` is public and returns `{"storageEnabled": false, "emailEnabled": false}` until build step 8 (spec 8).
- Frontend: server data lives only in TanStack Query; on logout or session expiry the whole query cache is cleared (spec 9).
- Tests never touch the real database (`tests/conftest.py` guard from step 3–4).
- New tables need RLS (not expected in this step; no migrations here).

```python
from functools import lru_cache

from pydantic import Field, field_validator
from pydantic_settings import BaseSettings, SettingsConfigDict


class Settings(BaseSettings):
    """Every setting Owl needs, read from environment variables or the .env file."""

    model_config = SettingsConfigDict(env_file=".env", extra="ignore")

    database_url: str
    # Signs login tokens. Anyone who knows it could forge a login, so it is long, random and secret.
    jwt_secret: str = Field(min_length=32)
    # Where the frontend runs. Used for the CSRF check and to decide if cookies need HTTPS.
    frontend_url: str = "http://localhost:5173"

    @field_validator("database_url")
    @classmethod
    def use_psycopg_driver(cls, value: str) -> str:
        # Supabase gives "postgresql://..."; SQLAlchemy needs to know which driver to use.
        for prefix in ("postgresql://", "postgres://"):
            if value.startswith(prefix):
                return "postgresql+psycopg://" + value.removeprefix(prefix)
        return value

    @field_validator("frontend_url")
    @classmethod
    def remove_trailing_slash(cls, value: str) -> str:
        return value.rstrip("/")

    @property
    def cookie_secure(self) -> bool:
        return self.frontend_url.startswith("https://")


@lru_cache
def get_settings() -> Settings:
    return Settings()
```

```python
MIN_PASSWORD_LENGTH = 10
MAX_PASSWORD_LENGTH = 128

# pwdlib's recommended settings use Argon2: deliberately slow, so guessing passwords is expensive.
_hasher = PasswordHash.recommended()

# Checked against when an email doesn't exist, so "unknown email" takes as long as "wrong password".

def hash_password(password: str) -> str:
    return _hasher.hash(password)


def verify_password(password: str, password_hash: str) -> bool:
    return _hasher.verify(password, password_hash)
```

Create `backend/app/auth/tokens.py`:

```python
import uuid
from dataclasses import dataclass
from datetime import UTC, datetime, timedelta

import jwt

from app.config import get_settings

SESSION_LIFETIME = timedelta(days=7)
ALGORITHM = "HS256"


@dataclass(frozen=True)
class SessionClaims:
    """What a valid login token tells us: who it is, and which login 'generation' it belongs to."""

    user_id: uuid.UUID
    token_version: int


def create_session_token(user_id: uuid.UUID, token_version: int, now: datetime | None = None) -> str:
    issued_at = now or datetime.now(UTC)
    payload = {
        "sub": str(user_id),
        "ver": token_version,
        "iat": issued_at,
        "exp": issued_at + SESSION_LIFETIME,
    }
    return jwt.encode(payload, get_settings().jwt_secret, algorithm=ALGORITHM)


def read_session_token(token: str) -> SessionClaims | None:
    """The claims if the token is genuine and not expired, otherwise None."""
    try:
        payload = jwt.decode(token, get_settings().jwt_secret, algorithms=[ALGORITHM])
        return SessionClaims(user_id=uuid.UUID(payload["sub"]), token_version=int(payload["ver"]))
    except (jwt.InvalidTokenError, KeyError, ValueError):
        return None
```

- [ ] **Step 3: Write the rate limiter**

Create `backend/app/auth/rate_limit.py`:

```python
import time
from collections import defaultdict, deque
from collections.abc import Callable


class RateLimiter:
    """Allows at most `max_attempts` per `window_seconds` for each key (an IP address or an email).

    Kept in memory, which is correct while only one backend runs (spec section 13).
    """

    def __init__(self, max_attempts: int, window_seconds: float, clock: Callable[[], float] = time.monotonic) -> None:
        self.max_attempts = max_attempts
        self.window_seconds = window_seconds
        self._clock = clock
        self._attempts: dict[str, deque[float]] = defaultdict(deque)

    def allow(self, key: str) -> bool:
        now = self._clock()
        attempts = self._attempts[key]
        # Forget attempts that are older than the window.
        while attempts and now - attempts[0] >= self.window_seconds:
            attempts.popleft()
        if len(attempts) >= self.max_attempts:
            return False
        attempts.append(now)
        return True

    def reset(self) -> None:
        self._attempts.clear()


login_by_ip = RateLimiter(max_attempts=20, window_seconds=15 * 60)
login_by_email = RateLimiter(max_attempts=10, window_seconds=15 * 60)
signup_by_ip = RateLimiter(max_attempts=10, window_seconds=60 * 60)


def reset_all_limiters() -> None:
    for limiter in (login_by_ip, login_by_email, signup_by_ip):
        limiter.reset()
```

Create `backend/app/auth/csrf.py`:

```python
from fastapi import Request
from fastapi.responses import JSONResponse

from app.config import get_settings

SAFE_METHODS = {"GET", "HEAD", "OPTIONS"}


async def require_same_origin(request: Request, call_next):
    """Block changes (POST, DELETE, ...) unless the browser says the request came from Owl's own page.

    Browsers add the Origin header themselves and pages can't fake it, so another website
    can't make a logged-in user's browser upload, delete or chat on their behalf.
    """
    if request.method not in SAFE_METHODS and request.headers.get("origin") != get_settings().frontend_url:
        return JSONResponse({"detail": "Request blocked: it didn't come from Owl."}, status_code=403)
    return await call_next(request)
```

```python
from fastapi import APIRouter, HTTPException, Request, Response, status
from sqlalchemy import select
from sqlalchemy.exc import IntegrityError

from app.auth.dependencies import SESSION_COOKIE, CurrentUser, DbSession
from app.auth.passwords import DUMMY_HASH, hash_password, verify_password
from app.auth.rate_limit import login_by_email, login_by_ip, signup_by_ip
from app.auth.schemas import LoginRequest, SignupRequest, UserOut
from app.auth.tokens import SESSION_LIFETIME, create_session_token
from app.config import get_settings
from app.models import User

router = APIRouter(prefix="/api/auth", tags=["auth"])


def client_ip(request: Request) -> str:
    return request.client.host if request.client else "unknown"


def too_many_attempts() -> HTTPException:
    return HTTPException(status.HTTP_429_TOO_MANY_REQUESTS, "Too many attempts. Please wait a few minutes and try again.")


def set_session_cookie(response: Response, user: User) -> None:
    response.set_cookie(
        SESSION_COOKIE,
        create_session_token(user.id, user.token_version),
        max_age=int(SESSION_LIFETIME.total_seconds()),
        httponly=True,
        secure=get_settings().cookie_secure,
        samesite="lax",
        path="/",
    )


@router.post("/signup", status_code=status.HTTP_201_CREATED)
def signup(body: SignupRequest, request: Request, response: Response, db: DbSession) -> UserOut:
    if not signup_by_ip.allow(client_ip(request)):
        raise too_many_attempts()
    user = User(email=body.email, password_hash=hash_password(body.password))
    db.add(user)
    try:
        db.commit()
    except IntegrityError:
        db.rollback()
        raise HTTPException(status.HTTP_409_CONFLICT, "An account with this email already exists.")
    set_session_cookie(response, user)
    return UserOut.model_validate(user)


@router.post("/login")
def login(body: LoginRequest, request: Request, response: Response, db: DbSession) -> UserOut:
    if not (login_by_ip.allow(client_ip(request)) and login_by_email.allow(body.email)):
        raise too_many_attempts()
    user = db.scalar(select(User).where(User.email == body.email))
    # Always check a hash, even for unknown emails, so both cases take the same time.
    password_ok = verify_password(body.password, user.password_hash if user else DUMMY_HASH)
    if user is None or not password_ok:
        raise HTTPException(status.HTTP_401_UNAUTHORIZED, "Wrong email or password.")
    set_session_cookie(response, user)
    return UserOut.model_validate(user)


@router.post("/logout", status_code=status.HTTP_204_NO_CONTENT)
def logout(response: Response) -> None:
    response.delete_cookie(
        SESSION_COOKIE, path="/", httponly=True, secure=get_settings().cookie_secure, samesite="lax"
    )


@router.get("/me")
def me(user: CurrentUser) -> UserOut:
    return UserOut.model_validate(user)
```

```python
class UnknownUserError(LookupError):
    """No account has that email."""


def reset_password(session: Session, email: str, new_password: str) -> None:
    """Set a new password and log the user out on every device."""
    if not MIN_PASSWORD_LENGTH <= len(new_password) <= MAX_PASSWORD_LENGTH:
        raise ValueError(f"Passwords must be {MIN_PASSWORD_LENGTH}–{MAX_PASSWORD_LENGTH} characters.")
    user = session.scalar(select(User).where(User.email == normalize_email(email)))

    user.password_hash = hash_password(new_password)
    user.token_version += 1
    session.commit()
```

```ts
export class ApiError extends Error {
  readonly status: number

  constructor(status: number, message: string) {
    super(message)
    this.name = "ApiError"
    this.status = status
  }
}

type RequestOptions = {
  method?: string
  body?: unknown
}

function parseJson(text: string): unknown {
  try {
    return JSON.parse(text)
  } catch {
    return null
  }
}

function errorMessage(data: unknown, status: number): string {
  if (typeof data === "object" && data !== null && "detail" in data && typeof data.detail === "string") {
    return data.detail
  }
  if (status === 422) return "Please check the form and try again."
  return "Something went wrong. Please try again."
}

/** Calls Owl's backend. `path` is relative to /api, e.g. "/auth/me". */
export async function apiFetch<T>(path: string, options: RequestOptions = {}): Promise<T> {
  const hasBody = options.body !== undefined
  const response = await fetch(`/api${path}`, {
    method: options.method ?? "GET",
    credentials: "same-origin",
    headers: hasBody ? { "Content-Type": "application/json" } : undefined,
    body: hasBody ? JSON.stringify(options.body) : undefined,
  })
  const text = await response.text()
  const data = text ? parseJson(text) : null
  if (!response.ok) throw new ApiError(response.status, errorMessage(data, response.status))
  return data as T
}
```

```ts
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query"
import { useNavigate } from "react-router"
import { ApiError, apiFetch } from "@/lib/api"
import type { User } from "@/lib/apiTypes"
import { CURRENT_USER_KEY } from "@/lib/queryClient"
import type { LoginValues, SignupValues } from "@/lib/validation"

/** The logged-in user, or null when nobody is logged in. */
export function useCurrentUser() {
  return useQuery({
    queryKey: CURRENT_USER_KEY,
    queryFn: async () => {
      try {
        return await apiFetch<User>("/auth/me")
      } catch (error) {
        if (error instanceof ApiError && error.status === 401) return null
        throw error
      }
    },
    staleTime: Infinity,
  })
}

export function useLogin() {
  const queryClient = useQueryClient()
  const navigate = useNavigate()
  return useMutation({
    mutationFn: (values: LoginValues) => apiFetch<User>("/auth/login", { method: "POST", body: values }),
    onSuccess: (user) => {
      queryClient.setQueryData(CURRENT_USER_KEY, user)
      navigate("/chat", { replace: true })
    },
  })
}

export function useSignup() {
  const queryClient = useQueryClient()
  const navigate = useNavigate()
  return useMutation({
    mutationFn: ({ email, password }: SignupValues) =>
      apiFetch<User>("/auth/signup", { method: "POST", body: { email, password } }),
    onSuccess: (user) => {
      queryClient.setQueryData(CURRENT_USER_KEY, user)
      navigate("/chat", { replace: true })
    },
  })
}

export function useLogout() {
  const queryClient = useQueryClient()
  const navigate = useNavigate()
  return useMutation({
    mutationFn: () => apiFetch<null>("/auth/logout", { method: "POST" }),
    onSettled: () => {
      // Forget every piece of this user's data so the next person on this browser sees none of it.
      queryClient.clear()
      queryClient.setQueryData(CURRENT_USER_KEY, null)
      navigate("/login", { replace: true })
    },
  })
}
```

```tsx
import { Loader2 } from "lucide-react"
import type { ReactNode } from "react"
import { Navigate, Outlet } from "react-router"
import { Button } from "@/components/ui/button"
import { useCurrentUser } from "@/hooks/useAuth"

function FullPageStatus({ children }: { children: ReactNode }) {
  return <div className="flex min-h-svh flex-col items-center justify-center gap-3 p-4 text-center">{children}</div>
}

function CannotReachServer({ onRetry }: { onRetry: () => void }) {
  return (
    <FullPageStatus>
      <p className="text-muted-foreground">Owl can't reach its server right now.</p>
      <Button variant="outline" onClick={onRetry}>
        Try again
      </Button>
    </FullPageStatus>
  )
}

/** Shows the page only to logged-in users; everyone else goes to /login. */
export function RequireAuth() {
  const { data: user, isPending, isError, refetch } = useCurrentUser()

  if (isPending) {
    return (
      <FullPageStatus>
        <Loader2 className="size-6 animate-spin text-muted-foreground" aria-label="Loading" />
      </FullPageStatus>
    )
  }
  if (isError) return <CannotReachServer onRetry={() => void refetch()} />
  if (!user) return <Navigate to="/login" replace />
  return <Outlet />
}

/** Login and sign-up pages: logged-in users go straight to the chat. */
export function GuestOnly() {
  const { data: user, isPending, isError, refetch } = useCurrentUser()

  if (isPending) return null
  if (isError) return <CannotReachServer onRetry={() => void refetch()} />
  if (user) return <Navigate to="/chat" replace />
  return <Outlet />
}
```
